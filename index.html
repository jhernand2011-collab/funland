  st={power:100,minute:0,activeCam:'cam1',doors:{l:false,r:false},flashCool:{l:false,r:false},phase:'idle',currentHour:0,

    entities:ENTITY_DEFS.map(e=>({...e,room:e.start,aggression:0,tickCount:0,active:false,retreatTimer:null}))};

  applyHour(0,true);

}


function applyHour(h,silent){

  const hdef=HOURS[h];st.currentHour=h;

  st.entities.forEach(e=>{

    if(hdef.active.includes(e.key)&&!e.active){e.active=true;e.room=e.start;e.aggression=0;}

    e.speed=hdef.speed[e.key]||999;

  });

  if(!silent&&h>=4)st.entities.forEach(e=>{if(e.active)e.aggression=Math.min(10,e.aggression+2);});

  document.getElementById('phase-label').textContent=hdef.label+' — '+hdef.name;

  document.getElementById('hour-badge').textContent=hdef.label;

  renderHourBar();renderLegend();

  if(!silent){const hs=HOUR_STORIES[h];if(hs)setStoryPanel(hs.tag,hs.text,hs.clues);sndStory();if(h>=3)playNoise(0.4,0.1);}

}


function startGame(){

  initState();

  document.getElementById('overlay').style.display='none';

  const m=document.querySelector('.story-modal');if(m)m.remove();

  renderHourBar();renderLegend();renderGrid();renderTracker();

  switchCam('cam1',true);updateHUD();

  const hs=HOUR_STORIES[0];setStoryPanel(hs.tag,hs.text,hs.clues);

  if(gameLoop)clearInterval(gameLoop);

  gameLoop=setInterval(tick,1000);

  if(!ambDone){ambDone=true;setTimeout(sndAmbient,2500);}

  setStatus('12:00 AM. ','You are alone in the park.');

}


function tick(){

  if(st.phase==='over')return;

  st.phase='playing';

  const hdef=HOURS[st.currentHour];

  let drain=hdef.drain;

  if(st.doors.l)drain+=0.20;

  if(st.doors.r)drain+=0.20;

  const op=st.entities.find(e=>e.key==='operator'&&e.active);

  if(op&&['cam1','cam2','cam5'].includes(op.room))drain+=0.14;

  st.power=Math.max(0,st.power-drain);

  st.minute++;

  const newHour=Math.floor(st.minute/60);

  if(newHour!==st.currentHour&&newHour<6)applyHour(newHour,false);

  if(st.minute%hdef.aggressionRate===0)st.entities.forEach(e=>{if(e.active)e.aggression=Math.min(10,e.aggression+1);});

  st.entities.forEach(e=>{

    if(!e.active)return;

    e.tickCount++;

    if(e.tickCount>=e.speed){e.tickCount=0;if(e.room!==e.key+'-approach')moveEntity(e);}

  });

  if(st.power<=0){lose('powerout');return;}

  if(st.minute>=360){win();return;}

  if(st.power<25&&st.minute%18===0)sndPowerLow();

  updateHUD();renderTracker();

}


function moveEntity(e){

  if(e.room===e.key+'-approach')return;

  if(Math.random()*10>10-e.aggression)return;

  const room=ROOMS.find(r=>r.id===e.room);if(!room)return;

  if(e.key==='operator'){const next=room.adj[Math.floor(Math.random()*room.adj.length)];e.room=next;if(st.activeCam===next)updateCamView();renderGrid();return;}

  const adj=room.adj;let next=adj[Math.floor(Math.random()*adj.length)];

  if(Math.random()<0.45&&adj.includes(e.via))next=e.via;

  if(next===e.via){

    e.room=e.key+'-approach';

    setStatus(e.name+' at the '+(e.side==='l'?'LEFT':'RIGHT')+' door — ','seal it now!');

    sndAlert(e.side==='l'?370:510);

    renderGrid();renderTracker();scheduleRetreat(e);return;

  }

  e.room=next;renderGrid();if(st.activeCam===next)updateCamView();

}


function scheduleRetreat(e){

  if(e.retreatTimer)clearTimeout(e.retreatTimer);

  e.retreatTimer=setTimeout(()=>{

    if(st.phase==='over')return;

    if(e.room===e.key+'-approach'){

      if(st.doors[e.side]){e.room='cam6';setStatus(e.name+' retreated. ','The door held.');renderGrid();renderTracker();}

      else lose('door');

    }

  },3500);

}


function toggleDoor(side){

  if(st.phase!=='playing')return;

  st.doors[side]=!st.doors[side];

  st.doors[side]?sndSeal():sndOpen();

  document.getElementById('door-'+side).classList.toggle('sealed',st.doors[side]);

  document.getElementById('d'+side+'state').textContent=st.doors[side]?'sealed':'open';

  document.getElementById('di-'+side).classList.toggle('sealed',st.doors[side]);

  st.power=Math.max(0,st.power-1.0);updateHUD();

}


function flashLight(side){

  if(st.flashCool[side]||st.phase!=='playing')return;

  sndFlash();st.power=Math.max(0,st.power-1.5);

  st.flashCool[side]=true;document.getElementById('flash-'+side).classList.add('cooldown');

  const fo=document.getElementById('flash-overlay');fo.classList.add('on');setTimeout(()=>fo.classList.remove('on'),70);

  const ap=st.entities.filter(e=>e.active&&e.side===side&&e.room===e.key+'-approach');

  if(ap.length){if(!st.doors[side]){setStatus(ap.map(e=>e.name).join(' & ')+' right there — ','SEAL THE DOOR!');sndAlert();}else setStatus('Flash: door sealed. ','Still waiting.');}

  else setStatus('Blind spot clear.','');

  updateHUD();setTimeout(()=>{st.flashCool[side]=false;document.getElementById('flash-'+side).classList.remove('cooldown');},1800);

}


function switchCam(id,force){

  if(!force)sndCam();

  document.getElementById('cam-view').classList.add('switching');

  st.power=Math.max(0,st.power-0.3);

  setTimeout(()=>{document.getElementById('cam-view').classList.remove('switching');st.activeCam=id;updateCamView();renderGrid();},110);

}


function updateCamView(){

  const room=ROOMS.find(r=>r.id===st.activeCam);if(!room)return;

  document.getElementById('cam-label').textContent=room.label;

  document.getElementById('room-art').textContent=room.icon;

  const wrap=document.getElementById('sprites-wrap');wrap.innerHTML='';

  let off=0;

  st.entities.forEach((e,i)=>{

    if(!e.active)return;

    const here=e.room===st.activeCam,approach=e.via===st.activeCam&&e.room===e.key+'-approach';

    if(here||approach){

      const el=document.createElement('div');el.className='esp '+(approach?'approach':'show');

      el.style.right=(10+off*46)+'px';

      el.style.filter='brightness(0.25) sepia(1) hue-rotate('+[280,120,40,0][i]+'deg)';

      el.textContent=e.emoji;wrap.appendChild(el);off++;

    }

  });

}


function renderGrid(){

  const grid=document.getElementById('cam-grid');grid.innerHTML='';

  ROOMS.forEach(r=>{

    const btn=document.createElement('div');

    let ac='';

    st.entities.forEach(e=>{if(!e.active)return;if(e.room===r.id||(e.via===r.id&&e.room===e.key+'-approach'))ac+=' '+e.ac;});

    btn.className='cam-btn'+(st.activeCam===r.id?' active':'')+ac;

    btn.innerHTML='<span style="display:inline-block;width:5px;height:5px;border-radius:50%;background:#1e1e1e;margin-right:2px;vertical-align:middle"></span>'+r.label.split('—')[0].trim()+'<br><span style="font-size:8px;color:#252525">'+r.label.split('—')[1].trim()+'</span>';

    btn.onclick=()=>switchCam(r.id);grid.appendChild(btn);

  });

}


function renderTracker(){

  const el=document.getElementById('et-row');el.innerHTML='';

  st.entities.forEach(e=>{

    const card=document.createElement('div');card.className='et-card';

    const atDoor=e.room===e.key+'-approach';

    const loc=!e.active?'sleeping':(atDoor?'AT DOOR!':(ROOMS.find(r=>r.id===e.room)?.label.split('—')[1].trim()||'—'));

    const bars='█'.repeat(Math.round(e.aggression/2))+'░'.repeat(5-Math.round(e.aggression/2));

    card.innerHTML=`<div class="et-name" style="color:${e.active?e.color:'#2a2a2a'}">${e.emoji} ${e.name}</div><div class="et-loc${atDoor?' danger':''}">${loc}</div><div class="et-agg" style="color:${e.active?e.color:'#2a2a2a'};opacity:0.5">${bars}</div>`;

    el.appendChild(card);

  });

}


function renderLegend(){

  const el=document.getElementById('legend');el.innerHTML='';

  st.entities.forEach(e=>{

    if(!e.active)return;

    const d=document.createElement('div');d.className='legend-item';

    d.innerHTML=`<div class="legend-dot" style="background:${e.color}"></div>${e.name}`;el.appendChild(d);

  });

}


function renderHourBar(){

  const bar=document.getElementById('hour-bar');bar.innerHTML='';

  HOURS.forEach((h,i)=>{

    const done=i<st.currentHour,cur=i===st.currentHour,danger=cur&&i>=4;

    const pip=document.createElement('div');pip.className='hour-pip';

    const dot=document.createElement('div');

    dot.className='hour-pip-dot'+(done?' done':'')+(cur&&!danger?' current':'')+(danger?' danger':'');

    dot.textContent=h.name;

    const lbl=document.createElement('div');lbl.className='hour-pip-label';lbl.textContent=h.label;

    pip.appendChild(dot);pip.appendChild(lbl);bar.appendChild(pip);

  });

}


function setStoryPanel(tag,text,clues){

  const panel=document.getElementById('story-panel');

  panel.className='story-panel new';setTimeout(()=>panel.classList.remove('new'),2500);

  panel.querySelector('.story-tag').textContent=tag;

  const textEl=document.getElementById('story-text');

  textEl.innerHTML=text.replace(/\n/g,'<br>');

  if(clues&&clues.length){

    const wrap=document.createElement('div');wrap.style.marginTop='6px';

    clues.forEach(c=>{const doc=DOCS[c];if(!doc)return;const btn=document.createElement('span');btn.className='story-clue';btn.textContent='['+doc.tag+']';btn.onclick=()=>showDoc(c);wrap.appendChild(btn);});

    textEl.appendChild(wrap);

  }

}


function showDoc(key){

  const doc=DOCS[key];if(!doc)return;

  const modal=document.createElement('div');modal.className='story-modal';

  modal.innerHTML=`<div class="doc-label">RECOVERED DOCUMENT</div><div class="doc-title">${doc.title}</div><div class="doc-body">${doc.body.replace(/\n/g,'<br>')}</div><div class="doc-footer">${doc.footer}</div><button onclick="this.closest('.story-modal').remove()">CLOSE DOCUMENT</button>`;

  document.getElementById('g').appendChild(modal);

}


function setStatus(p,h){document.getElementById('status').innerHTML=p+(h?'<span>'+h+'</span>':'');}


function updateHUD(){

  const pct=Math.max(0,Math.round(st.power));

  document.getElementById('power-txt').textContent=pct+'%';

  document.getElementById('power-txt').className='power'+(pct<25?' low':'');

  const bar=document.getElementById('power-bar');bar.style.width=pct+'%';

  bar.style.background=pct<25?'#e05050':pct<50?'#c08020':'#7ec87e';

  const h=Math.floor(st.minute/60),m=st.minute%60,hr=h===0?12:h>12?h-12:h;

  document.getElementById('clock').textContent=hr+':'+String(m).padStart(2,'0')+' '+(h<12?'AM':'PM');

}


function lose(type){

  if(st.phase==='over')return;

  st.phase='over';clearInterval(gameLoop);sndJumpscare();

  const s=document.getElementById('scare');s.style.opacity=0.85;

  document.getElementById('g').classList.add('shake');

  setTimeout(()=>{

    s.style.opacity=0;document.getElementById('g').classList.remove('shake');

    const h=HOURS[st.currentHour];

    const bodies={

      powerout:'The monitors went dark one by one.\n\nIn the silence you heard them moving — all of them, from every direction at once. The door seals disengaged when power failed.\n\nYou had nowhere to go.',

      door:"You saw it on the monitor. You waited one second too long.\n\nThe booth door opened from the outside. That shouldn't be possible — the booth locks from within.\n\nIt was possible.",

    };

    showOverlay('YOU DIDN\'T MAKE IT','#e05050',

      `<p style="color:#444;font-size:9px;text-transform:uppercase;letter-spacing:0.08em;margin-bottom:8px">Security Log — Terminated at ${h.label}</p>`+

      `<p style="text-align:left;color:#666;font-size:11px;line-height:1.85;max-width:340px">${(bodies[type]||bodies.door).replace(/\n/g,'<br>')}</p>`+

      `<p style="color:#333;font-size:9px;margin-top:8px;text-transform:uppercase;letter-spacing:0.08em">Report filed: INCOMPLETE</p>`,

      'CLOCK IN AGAIN');

  },700);

}


function win(){

  if(st.phase==='over')return;

  st.phase='over';clearInterval(gameLoop);sndWin();

  showOverlay('6:00 AM','#f0c040',

    `<p style="color:#5a9a2a;font-size:9px;text-transform:uppercase;letter-spacing:0.1em;margin-bottom:8px">Security Report — Filed Successfully</p>`+

    `<p style="text-align:left;color:#666;font-size:11px;line-height:1.85;max-width:340px">You stayed the whole night.<br><br>The entities retreated at first light. The carousel slowed. The horses settled into stillness.<br><br><em style="color:#a0a040">You write in your report: "Nothing unusual observed."</em><br><br>In a drawer you find a new entry in handwriting you don't recognize:<br><br><em style="color:#a0a040">"Good. You're learning. Come back tomorrow."</em></p>`+

    `<p style="color:#333;font-size:9px;margin-top:8px;text-transform:uppercase;letter-spacing:0.08em">Report filed: 6:01 AM — Shift complete</p>`,

    'CLOCK IN AGAIN');

}


function showOverlay(title,color,bodyHTML,btnLabel){

  const ov=document.getElementById('overlay');

  ov.innerHTML=`<h2 style="color:${color};letter-spacing:0.1em;margin-bottom:6px">${title}</h2>${bodyHTML}<button class="primary" onclick="startGame()" style="margin-top:12px">${btnLabel}</button>`;

  ov.style.display='flex';

}
